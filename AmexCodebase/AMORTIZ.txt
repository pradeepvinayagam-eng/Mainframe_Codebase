       IDENTIFICATION DIVISION.
       PROGRAM-ID. AMORTIZ.
      *****************************************************************
      * AMORTIZATION CALCULATION SUBPROGRAM                          *
      *****************************************************************
       DATA DIVISION.
       WORKING-STORAGE SECTION.
       01  WS-MONTHLY-RATE             PIC 9V9(8) COMP-3.
       01  WS-TEMP-CALC                PIC S9(13)V99 COMP-3.

       LINKAGE SECTION.
       01  LS-LOAN-NUMBER              PIC X(20).
       01  LS-PRINCIPAL-BALANCE        PIC S9(13)V99 COMP-3.
       01  LS-INTEREST-RATE            PIC 9V9(6) COMP-3.
       01  LS-REMAINING-TERM           PIC 9(5) COMP.
       01  LS-SCHEDULED-PAYMENT        PIC S9(9)V99 COMP-3.
       01  LS-CALCULATION-STATUS       PIC X.
       01  LS-RETURN-CODE              PIC S9(4) COMP.

       PROCEDURE DIVISION USING LS-LOAN-NUMBER
                                LS-PRINCIPAL-BALANCE
                                LS-INTEREST-RATE
                                LS-REMAINING-TERM
                                LS-SCHEDULED-PAYMENT
                                LS-CALCULATION-STATUS
                                LS-RETURN-CODE.
       0000-MAIN.
           MOVE 'Y' TO LS-CALCULATION-STATUS
           COMPUTE WS-MONTHLY-RATE = LS-INTEREST-RATE / 12
           COMPUTE LS-SCHEDULED-PAYMENT ROUNDED =
               LS-PRINCIPAL-BALANCE *
               (WS-MONTHLY-RATE * (1 + WS-MONTHLY-RATE) **
                LS-REMAINING-TERM) /
               (((1 + WS-MONTHLY-RATE) ** LS-REMAINING-TERM) - 1)
           GOBACK.
